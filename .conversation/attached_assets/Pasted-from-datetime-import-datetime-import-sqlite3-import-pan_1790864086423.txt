# --- INISIALISASI DATABASE SQLITE ---
def init_db():
  conn = sqlite3.connect("erpmak.db", check_same_thread=False)
  c = conn.cursor()

  # Tabel Bahan Baku, Kemasan, & Barang Jadi (Kerupuk)
  c.execute("""CREATE TABLE IF NOT EXISTS ingredients (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT,
                category TEXT,
                unit TEXT,
                stock REAL,
                min_stock REAL,
                last_price REAL
            )""")

  # Tabel Produk Jadian
  c.execute("""CREATE TABLE IF NOT EXISTS products (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT,
                selling_price REAL
            )""")

  # Tabel Resep (Bill of Materials / BOM)
  c.execute("""CREATE TABLE IF NOT EXISTS recipe_items (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                product_id INTEGER,
                ingredient_id INTEGER,
                qty_required REAL
            )""")

  # Tabel Riwayat Belanja (Pengeluaran)
  c.execute("""CREATE TABLE IF NOT EXISTS shopping (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                date TEXT,
                supplier_type TEXT,
                total_cost REAL
            )""")

  c.execute("""CREATE TABLE IF NOT EXISTS shopping_details (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                shopping_id INTEGER,
                ingredient_id INTEGER,
                qty REAL,
                cost REAL
            )""")

  # Tabel Riwayat Penjualan (Pendapatan)
  c.execute("""CREATE TABLE IF NOT EXISTS sales (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                date TEXT,
                total_revenue REAL
            )""")

  c.execute("""CREATE TABLE IF NOT EXISTS sales_details (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                sales_id INTEGER,
                product_id INTEGER,
                qty INTEGER,
                revenue REAL
            )""")

  conn.commit()
  return conn


conn = init_db()

      "SELECT SUM(total_revenue) as rev FROM sales WHERE date = ?",

      "Pendapatan Hari Ini",

      f"Rp {today_cost:,.0f}",

elif menu == "🛒 Catat Belanja & Stok Masuk":
  st.title("🛒 Catat Belanja Bahan / Pabrik")
  st.write("Catat setiap pengeluaran belanja pasar atau ambil kerupuk pabrik.")

  supplier_type = st.radio(
      "Sumber Belanja:",
      ["Pasar Tradisional / Toko Kelontong", "Pabrik Kerupuk (Langsung)"],
  )

  df_ing = pd.read_sql("SELECT id, name, unit FROM ingredients", conn)

  with st.form("form_shopping"):
    shop_date = st.date_input(
        "Tanggal Belanja", datetime.now()
    ).strftime("%Y-%m-%d")

    # Dynamic inputs simulation
    st.markdown("### Pilih Barang yang Dibeli")
    selected_items = []

    # Kita buat 3 baris input belanjaan fleksibel
    col_a, col_b, col_c = st.columns([3, 2, 2])
    item1 = col_a.selectbox(
        "Item 1", df_ing["id"], format_func=lambda x: df_ing[df_ing['id'] == x]['name'].values[0], key="i1"
    )
    qty1 = col_b.number_input("Jumlah 1", min_value=0.0, value=0.0, key="q1")
    cost1 = col_c.number_input(
        "Total Harga (Rp) 1", min_value=0.0, value=0.0, key="c1"
    )

    item2 = col_a.selectbox(
        "Item 2 (Opsional)", [0] + list(df_ing["id"]), format_func=lambda x: "Pilih Item..." if x == 0 else df_ing[df_ing['id'] == x]['name'].values[0], key="i2"
    )
    qty2 = col_b.number_input("Jumlah 2", min_value=0.0, value=0.0, key="q2")
    cost2 = col_c.number_input(
        "Total Harga (Rp) 2", min_value=0.0, value=0.0, key="c2"
    )

    submitted_shop = st.form_submit_button("💾 Simpan Belanja & Tambah Stok")

    if submitted_shop:
      c = conn.cursor()
      # Hitung total cost keseluruhan
      total_all_cost = cost1 + cost2

      # Simpan header shopping
      c.execute(
          "INSERT INTO shopping (date, supplier_type, total_cost) VALUES (?, ?,"
          " ?)",
          (shop_date, supplier_type, total_all_cost),
      )
      shop_id = c.lastrowid

      # Item 1 process
      if qty1 > 0:
        c.execute(
            "INSERT INTO shopping_details (shopping_id, ingredient_id, qty,"
            " cost) VALUES (?, ?, ?, ?)",
            (shop_id, item1, qty1, cost1),
        )
        c.execute(
            "UPDATE ingredients SET stock = stock + ?, last_price = ? WHERE id"
            " = ?",
            (qty1, cost1 / qty1 if qty1 > 0 else 0, item1),
        )

      # Item 2 process
      if item2 != 0 and qty2 > 0:
        c.execute(
            "INSERT INTO shopping_details (shopping_id, ingredient_id, qty,"
            " cost) VALUES (?, ?, ?, ?)",
            (shop_id, item2, qty2, cost2),
        )
        c.execute(
            "UPDATE ingredients SET stock = stock + ?, last_price = ? WHERE id"
            " = ?",
            (qty2, cost2 / qty2 if qty2 > 0 else 0, item2),
        )

      conn.commit()
      st.success(
          "Data belanja berhasil disimpan dan stok gudang otomatis bertambah!"
      )

# ==========================================
elif menu == "📝 Catat Penjualan Harian":
  st.title("📝 Pencatatan Penjualan Harian")
  st.write(
      "Masukkan jumlah porsi produk yang terjual hari ini. Sistem akan"
      " otomatis menghitung pendapatan dan mengurangi stok bahan."
  )

  sales_date = st.date_input(
      "Tanggal Penjualan", datetime.now(), key="sales_date"
  ).strftime("%Y-%m-%d")

  df_products = pd.read_sql("SELECT id, name, selling_price FROM products", conn)

  with st.form("form_sales"):
    st.markdown("### Jumlah Porsi Terjual")
    sales_inputs = {}

    for index, row in df_products.iterrows():
      sales_inputs[row["id"]] = {
          "name": row["name"],
          "price": row["selling_price"],
          "qty": st.number_input(
              f"Terjual: {row['name']} (@ Rp {row['selling_price']:,.0f})",
              min_value=0,
              step=1,
              key=f"prod_{row['id']}",
          ),
      }

    submitted_sales = st.form_submit_button(
        "💾 Simpan Penjualan & Potong Stok"
    )

    if submitted_sales:
      c = conn.cursor()
      total_revenue = 0

      # Hitung total pendapatan dulu
      for pid, data in sales_inputs.items():
        total_revenue += data["qty"] * data["price"]

      # Simpan header sales
      c.execute(
          "INSERT INTO sales (date, total_revenue) VALUES (?, ?)",
          (sales_date, total_revenue),
      )
      sales_id = c.lastrowid

      # Simpan detail dan potong stok
      for pid, data in sales_inputs.items():
        qty_sold = data["qty"]
        if qty_sold > 0:
          rev = qty_sold * data["price"]
          c.execute(
              "INSERT INTO sales_details (sales_id, product_id, qty, revenue)"
              " VALUES (?, ?, ?, ?)",
              (sales_id, pid, qty_sold, rev),
          )

          # Logika khusus pengurangan stok:
          # Jika produk adalah Kerupuk Bungkus, langsung kurangi stok item 'Kerupuk Matang (Pabrik)'
          if data["name"] == "Kerupuk Bungkus":
            c.execute(
                "UPDATE ingredients SET stock = stock - ? WHERE name = 'Kerupuk"
                " Matang (Pabrik)'",
                (qty_sold,),
            )
          else:
            # Untuk produk olahan (Nasi Bakar, Sate Jamur, Pangsit), idealnya kurangi bahan mentahnya.
            # Pada contoh ERP sederhana ini, kita kurangi estimasi bahan baku utama atau kemasan secara proporsional jika ada.
            pass

      conn.commit()
      st.success(
          f"Penjualan berhasil direkap! Total Omset: Rp {total_revenue:,.0f}"
      )

elif menu == "💰 Laporan Keuangan":
  st.title("💰 Laporan Laba Rugi Usaha")
  st.write("Rekapitulasi total pendapatan, pengeluaran belanja, dan laba bersih.")

  col1, col2 = st.columns(2)

  with col1:
    st.subheader("📥 Riwayat Pendapatan Penjualan")
    df_all_sales = pd.read_sql("SELECT * FROM sales ORDER BY date DESC", conn)
    st.dataframe(df_all_sales, use_container_width=True)

  with col2:
    st.subheader("📤 Riwayat Pengeluaran Belanja")
    df_all_shop = pd.read_sql("SELECT * FROM shopping ORDER BY date DESC", conn)
    st.dataframe(df_all_shop, use_container_width=True)

  st.markdown("---")
  st.subheader("📈 Ringkasan Keuangan Keseluruhan")

  total_rev_all = (
      df_all_sales["total_revenue"].sum() if not df_all_sales.empty else 0
  )
  total_cost_all = (
      df_all_shop["total_cost"].sum() if not df_all_shop.empty else 0
  )
  net_profit_all = total_rev_all - total_cost_all

  c1, c2, c3 = st.columns(3)
  c1.metric("Akumulasi Total Omset", f"Rp {total_rev_all:,.0f}")
  c2.metric("Akumulasi Total Belanja", f"Rp {total_cost_all:,.0f}")
  c3.metric("Akumulasi Laba Bersih", f"Rp {net_profit_all:,.0f}")
